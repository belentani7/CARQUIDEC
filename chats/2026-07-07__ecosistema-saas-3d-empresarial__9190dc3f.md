# services/cost_calculator.py
import numpy as np
from stl import mesh
import trimesh
from datetime import datetime
from config import Config

class PrintCostCalculator:
    """
    Calculadora profesional de costes de impresión 3D
    Considera: material, electricidad, depreciación, mano de obra, margen
    """
    
    def __init__(self):
        self.electricity_rate = Config.ELECTRICITY_RATE  # €/kWh
        self.material_prices = Config.DEFAULT_MATERIAL_PRICE
        
    def analyze_stl_file(self, file_path):
        """Analiza un archivo STL y extrae métricas"""
        try:
            # Cargar mesh
            loaded_mesh = mesh.Mesh.from_file(file_path)
            
            # Calcular volumen
            volume = loaded_mesh.get_volume()  # mm³
            volume_cm3 = volume / 1000  # cm³
            
            # Calcular área superficial
            surface_area = sum(loaded_mesh.areas)  # mm²
            surface_area_cm2 = surface_area / 100  # cm²
            
            # Bounding box
            min_x = loaded_mesh.x.min()
            max_x = loaded_mesh.x.max()
            min_y = loaded_mesh.y.min()
            max_y = loaded_mesh.y.max()
            min_z = loaded_mesh.z.min()
            max_z = loaded_mesh.z.max()
            
            bounding_box = {
                'x': (max_x - min_x) / 10,  # cm
                'y': (max_y - min_y) / 10,  # cm
                'z': (max_z - min_z) / 10   # cm
            }
            
            # Verificar si es manifold (sólido cerrado)
            is_manifold = self._check_manifold(loaded_mesh)
            
            # Detectar errores
            errors = self._detect_errors(loaded_mesh)
            
            return {
                'volume_cm3': volume_cm3,
                'surface_area_cm2': surface_area_cm2,
                'bounding_box': bounding_box,
                'is_manifold': is_manifold,
                'errors': errors,
                'triangle_count': len(loaded_mesh.vectors)
            }
            
        except Exception as e:
            return {'error': str(e)}
    
    def calculate_weight(self, volume_cm3, material='PLA', density=None):
        """Calcula el peso en gramos basado en el volumen y material"""
        densities = {
            'PLA': 1.24,      # g/cm³
            'ABS': 1.04,
            'PETG': 1.27,
            'RESINA': 1.10,
            'TPU': 1.20,
            'NYLON': 1.14
        }
        
        if density is None:
            density = densities.get(material, 1.24)
        
        return volume_cm3 * density
    
    def calculate_print_time(self, weight_grams, layer_height=0.2, 
                            infill=20, print_speed=50):
        """
        Calcula tiempo estimado de impresión
        Fórmula simplificada basada en peso y parámetros
        """
        # Tasa de extrusión aproximada (g/hora)
        # Depende de layer height, nozzle size, y print speed
        base_rate = 15  # g/hora para configuración estándar
        
        # Ajustar por layer height
        layer_factor = 0.2 / layer_height
        
        # Ajustar por infill
        infill_factor = 1 + (infill / 100) * 0.5
        
        # Ajustar por velocidad
        speed_factor = 50 / print_speed
        
        effective_rate = base_rate * layer_factor * infill_factor * speed_factor
        
        print_time_hours = weight_grams / effective_rate
        
        # Añadir tiempo para capas iniciales y travel
        print_time_hours *= 1.15
        
        return round(print_time_hours, 2)
    
    def calculate_cost(self, weight_grams, print_time_hours, material='PLA',
                      infill=20, layer_height=0.2, electricity_consumption=200,
                      labor_time_minutes=10, labor_rate=15, 
                      depreciation_rate=0.05, margin=30, printer_cost=500,
                      printer_hours_life=5000):
        """
        Calcula el coste completo y precio de venta
        
        Parámetros:
        - weight_grams: Peso de la pieza en gramos
        - print_time_hours: Tiempo de impresión en horas
        - material: Tipo de material (PLA, ABS, PETG, etc.)
        - infill: Porcentaje de relleno (0-100)
        - layer_height: Altura de capa en mm
        - electricity_consumption: Consumo eléctrico en watts
        - labor_time_minutes: Tiempo de mano de obra en minutos
        - labor_rate: Coste hora de mano de obra en €
        - depreciation_rate: Tasa de depreciación anual
        - margin: Margen de beneficio en %
        - printer_cost: Coste de la impresora en €
        - printer_hours_life: Vida útil estimada en horas
        
        Retorna:
        - Diccionario con desglose de costes y precio final
        """
        
        # 1. Coste de material
        material_price_per_kg = self.material_prices.get(material, 20.0)
        material_cost = (weight_grams / 1000) * material_price_per_kg
        
        # Añadir soporte y purga (estimado 10%)
        material_cost *= 1.10
        
        # 2. Coste eléctrico
        kwh_consumed = (electricity_consumption / 1000) * print_time_hours
        electricity_cost = kwh_consumed * self.electricity_rate
        
        # 3. Depreciación de la impresora
        depreciation_per_hour = printer_cost / printer_hours_life
        depreciation_cost = depreciation_per_hour * print_time_hours
        
        # 4. Coste de mano de obra
        labor_hours = labor_time_minutes / 60
        labor_cost = labor_hours * labor_rate
        
        # 5. Coste de mantenimiento (estimado 5% del coste de impresora/año)
        maintenance_cost = depreciation_cost * 0.1
        
        # COSTE TOTAL DE PRODUCCIÓN
        production_cost = (material_cost + electricity_cost + 
                          depreciation_cost + labor_cost + maintenance_cost)
        
        # 6. Margen de beneficio
        profit = production_cost * (margin / 100)
        
        # PRECIO DE VENTA
        sale_price = production_cost + profit
        
        # Precio por gramo
        price_per_gram = sale_price / weight_grams if weight_grams > 0 else 0
        
        return {
            'material_cost': round(material_cost, 2),
            'electricity_cost': round(electricity_cost, 2),
            'depreciation_cost': round(depreciation_cost, 2),
            'labor_cost': round(labor_cost, 2),
            'maintenance_cost': round(maintenance_cost, 2),
            'production_cost': round(production_cost, 2),
            'profit': round(profit, 2),
            'sale_price': round(sale_price, 2),
            'price_per_gram': round(price_per_gram, 2),
            'margin_percent': margin,
            'breakdown': {
                'weight_grams': round(weight_grams, 2),
                'print_time_hours': round(print_time_hours, 2),
                'material': material,
                'material_price_per_kg': material_price_per_kg,
                'kwh_consumed': round(kwh_consumed, 3),
                'labor_minutes': labor_time_minutes
            }
        }
    
    def _check_manifold(self, mesh_obj):
        """Verifica si el mesh es manifold (sólido cerrado)"""
        try:
            # Verificar que todas las aristas estén compartidas por exactamente 2 triángulos
            edges = {}
            for triangle in mesh_obj.vectors:
                for i in range(3):
                    edge = tuple(sorted([tuple(triangle[i]), tuple(triangle[(i+1)%3])]))
                    edges[edge] = edges.get(edge, 0) + 1
            
            # En un mesh manifold, cada arista debe aparecer exactamente 2 veces
            # (excepto en los bordes abiertos)
            non_manifold_edges = sum(1 for count in edges.values() if count != 2)
            
            return non_manifold_edges < len(edges) * 0.01  # 1% de tolerancia
            
        except:
            return False
    
    def _detect_errors(self, mesh_obj):
        """Detecta errores comunes en el mesh"""
        errors = []
        
        # 1. Triángulos invertidos (normales inconsistentes)
        if not np.all(np.isfinite(mesh_obj.vectors)):
            errors.append("Triángulos con valores infinitos o NaN")
        
        # 2. Triángulos degenerados (área cero)
        zero_area = np.sum(mesh_obj.areas == 0)
        if zero_area > 0:
            errors.append(f"{zero_area} triángulos degenerados (área cero)")
        
        # 3. Verificar volumen negativo (normales invertidas)
        try:
            volume = mesh_obj.get_volume()
            if volume < 0:
                errors.append("Volumen negativo - posibles normales invertidas")
        except:
            errors.append("No se puede calcular el volumen")
        
        return errors
    
    def recommend_orientation(self, bounding_box, surface_area):
        """Recomienda la mejor orientación de impresión"""
        # La mejor orientación minimiza:
        # 1. Altura Z (menos capas = más rápido)
        # 2. Área de soportes necesaria
        # 3. Maximiza área de base (mejor adherencia)
        
        dimensions = {
            'x': bounding_box['x'],
            'y': bounding_box['y'],
            'z': bounding_box['z']
        }
        
        sorted_dims = sorted(dimensions.items(), key=lambda x: x[1])
        
        recommendation = {
            'best_orientation': f"Rotar para que el eje {sorted_dims[2][0]} sea Z",
            'estimated_support_reduction': "20-40%",
            'reasoning': f"La dimensión más pequeña ({sorted_dims[0][0]}: {sorted_dims[0][1]:.1f}cm) debe ser la altura para minimizar tiempo"
        }
        
        return recommendation

    def _analyze_geometry(self, mesh) -> Dict:
        """Análisis geométrico detallado"""
        bounds = mesh.bounds
        dimensions = bounds[1] - bounds[0]
        
        # Calcular centro de masa
        try:
            center_of_mass = mesh.center_mass
        except:
            center_of_mass = mesh.centroid
        
        # Calcular momentos de inercia
        try:
            inertia = mesh.moment_inertia
        except:
            inertia = None
        
        return {
            'dimensions_mm': {
                'x': dimensions[0],

            },
            'bounding_box_cm': {
                'x': dimensions[0] / 10,

            },

        # Detectar pequeñas características

    def _detect_overhangs(self, mesh, angle_threshold=45) -> Dict:
        """Detecta áreas con overhang que necesitan soportes"""
        
        overhang_faces = []
        
        for i, face in enumerate(mesh.faces):
            normal = mesh.face_normals[i]
            # Si la normal apunta hacia abajo más del threshold
            if normal[2] < -np.cos(np.radians(angle_threshold)):
                overhang_faces.append(i)
        
        overhang_area = len(overhang_faces) / len(mesh.faces) * 100
        
        return {
            'percentage': round(overhang_area, 2),
            'needs_supports': overhang_area > 5,
            'severity': 'high' if overhang_area > 30 else 'medium' if overhang_area > 15 else 'low'
        }
    
    def _detect_bridges(self, mesh) -> Dict:
        """Detecta puentes que podrían colapsar"""
        # Simplificación: detectar caras horizontales superiores
        # sobre espacios vacíos
        
        bridge_faces = []
        for i, face in enumerate(mesh.faces):
            normal = mesh.face_normals[i]
            # Caras apuntando hacia arriba
            if normal[2] > 0.9:
                # Verificar si hay espacio vacío debajo
                bridge_faces.append(i)
        
        return {
            'count': len(bridge_faces),
            'max_span_mm': 0,  # Implementar detección de span
            'needs_cooling': len(bridge_faces) > 0
        }
    
    def _detect_thin_walls(self, mesh, min_thickness=0.8) -> Dict:
        """Detecta paredes más delgadas que el nozzle"""
        # Usar ray casting para detectar espesores
        thin_regions = []
        
        # Muestrear puntos en la superficie
        for _ in range(100):  # 100 muestras
            point, face_idx = mesh.sample(1, return_index=True)
            normal = mesh.face_normals[face_idx[0]]
            
            # Ray en dirección opuesta a la normal
            locations, index_ray, index_tri = mesh.ray.intersects_location(
                ray_origins=point,
                ray_directions=-normal
            )
            
            if len(locations) > 0:
                distance = np.linalg.norm(locations[0] - point[0])
                if distance < min_thickness:
                    thin_regions.append(distance)
        
        return {
            'detected': len(thin_regions) > 0,
            'count': len(thin_regions),
            'min_thickness_mm': min(thin_regions) * 10 if thin_regions else None,
            'recommendation': 'Aumentar espesor de paredes' if thin_regions else 'OK'
        }
    
    def _detect_small_features(self, mesh, min_size=2.0) -> Dict:
        """Detecta características pequeñas difíciles de imprimir"""
        # Detectar vértices aislados o aristas cortas
        small_features = []
        
        edges = mesh.edges_unique
        edge_lengths = mesh.edges_unique_length
        
        short_edges = edge_lengths[edge_lengths < min_size / 10]  # convertir a mm
        
        return {
            'short_edges_count': len(short_edges),
            'min_edge_length_mm': float(np.min(edge_lengths) * 10) if len(edge_lengths) > 0 else None,
            'printability': 'difficult' if len(short_edges) > 10 else 'normal'
        }
    
    def _detect_errors(self, mesh) -> List[Dict]:
        """Detecta errores en el mesh"""
        errors = []
        
        # 1. Mesh no watertight
        if not mesh.is_watertight:
            errors.append({
                'type': 'not_watertight',
                'severity': 'high',
                'message': 'El modelo no es watertight (tiene agujeros)',
                'fix': 'Usar reparación de mesh o cerrar agujeros en CAD'
            })
        
        # 2. Normales inconsistentes
        if not mesh.is_winding_consistent:
            errors.append({
                'type': 'inconsistent_normals',
                'severity': 'medium',
                'message': 'Normales de caras inconsistentes',
                'fix': 'Recalcular normales en el software de modelado'
            })
        
        # 3. Triángulos degenerados
        degenerate_faces = mesh.nondegenerate_faces()
        if len(degenerate_faces) < len(mesh.faces):
            count = len(mesh.faces) - len(degenerate_faces)
            errors.append({
                'type': 'degenerate_triangles',
                'severity': 'medium',
                'message': f'{count} triángulos degenerados detectados',
                'fix': 'Eliminar triángulos degenerados'
            })
        
        # 4. Caras duplicadas
        if hasattr(mesh, 'unique_faces'):
            unique = mesh.unique_faces()
            if len(unique) < len(mesh.faces):
                errors.append({
                    'type': 'duplicate_faces',
                    'severity': 'low',
                    'message': 'Caras duplicadas detectadas',
                    'fix': 'Eliminar caras duplicadas'
                })
        
        # 5. Volumen negativo
        try:
            if mesh.volume < 0:
                errors.append({
                    'type': 'negative_volume',
                    'severity': 'high',
                    'message': 'Volumen negativo - normales posiblemente invertidas',
                    'fix': 'Invertir normales del mesh'
                })
        except:
            pass
        
        return errors
    
    def _suggest_optimizations(self, mesh) -> Dict:
        """Sugiere optimizaciones para el modelo"""
        suggestions = []
        
        # 1. Orientación óptima
        best_orientation = self._find_best_orientation(mesh)
        suggestions.append({
            'type': 'orientation',
            'suggestion': f"Orientar con eje {best_orientation['axis']} como Z",
            'benefit': f"Reducción de soportes: {best_orientation['support_reduction']}%",
            'priority': 'high'
        })
        
        # 2. Reducción de soportes
        overhangs = self._detect_overhangs(mesh)
        if overhangs['percentage'] > 20:
            suggestions.append({
                'type': 'support_reduction',
                'suggestion': 'Considerar dividir el modelo en partes',
                'benefit': 'Menor uso de material de soporte',
                'priority': 'medium'
            })
        
        # 3. Hollowing (hacer hueco)
        volume = mesh.volume / 1000 if mesh.is_watertight else 0
        if volume > 50:  # Más de 50 cm³
            suggestions.append({
                'type': 'hollowing',
                'suggestion': 'Considerar imprimir hueco con 2-3 perímetros',
                'benefit': f"Ahorro estimado: {volume * 0.8 * 1.24:.1f}g de material",
                'priority': 'high'
            })
        
        return {
            'suggestions': suggestions,
            'estimated_time_savings': self._estimate_time_savings(suggestions, mesh)
        }
    
    def _find_best_orientation(self, mesh) -> Dict:
        """Encuentra la mejor orientación de impresión"""
        dimensions = mesh.bounds[1] - mesh.bounds[0]
        
        # Probar las 6 orientaciones posibles
        orientations = [
            {'axis': 'X', dimension': dimensions[0], 'height': max(dimensions[1], dimensions[2])},
            {'axis': 'Y', 'dimension': dimensions[1], 'height': max(dimensions[0], dimensions[2])},
            {'axis': 'Z', 'dimension': dimensions[2], 'height': max(dimensions[0], dimensions[1])}
        ]
        
        # La mejor es la que minimiza la altura (menos capas)
        best = min(orientations, key=lambda x: x['height'])
        
        return {
            'axis': best['axis'],
            'height_mm': best['height'],
            'support_reduction': 30  # Estimado
        }
    
    def _estimate_cost_parameters(self, mesh) -> Dict:
        """Estima parámetros para cálculo de costes"""
        volume = mesh.volume / 1000 if mesh.is_watertight else 0
        
        # Peso estimado (PLA)
        weight = volume * 1.24  # densidad PLA
        
        # Añadir 15% para soportes
        weight_with_supports = weight * 1.15
        
        # Tiempo estimado (simplificado)
        # Asumir 15g/hora como tasa base
        print_time_hours = weight_with_supports / 15
        
        return {
            'volume_cm3': round(volume, 2),
            'weight_pla_grams': round(weight, 2),
            'weight_with_supports_grams': round(weight_with_supports, 2),
            'estimated_print_time_hours': round(print_time_hours, 2),
            'layer_count_0.2mm': round((mesh.bounds[1][2] - mesh.bounds[0][2]) / 0.2) if mesh.is_watertight else 0
        }
    
    def _calculate_aspect_ratio(self, dimensions) -> float:
        """Calcula el ratio de aspecto"""
        return max(dimensions) / min(dimensions) if min(dimensions) > 0 else 0
    
    def _calculate_printability_score(self, overhangs, bridges, thin_walls, 
                                     small_features) -> int:
        """Calcula un score de imprimibilidad (0-100)"""
        score = 100
        
        # Penalizar overhangs
        score -= overhangs['percentage'] * 0.5
        
        # Penalizar puentes grandes
        score -= bridges['count'] * 2
        
        # Penalizar paredes delgadas
        if thin_walls['detected']:
            score -= 20
        
        # Penalizar características pequeñas
        if small_features['printability'] == 'difficult':
            score -= 15
        
        return max(0, min(100, int(score)))
    
    def _generate_recommendations(self, analysis) -> List[str]:
        """Genera recomendaciones basadas en el análisis"""
        recommendations = []
        
        # Basado en printabilidad
        if analysis['printability']['overall_score'] < 60:
            recommendations.append("⚠️ El modelo tiene problemas de imprimibilidad. Considere modificar el diseño.")
        
        # Basado en overhangs
        if analysis['printability']['overhangs']['needs_supports']:
            recommendations.append("📌 Se requieren soportes. Use soportes en árbol para mejor calidad.")
        
        # Basado en errores
        if analysis['errors']:
            recommendations.append("🔧 Repare los errores del mesh antes de imprimir.")
        
        # Basado en optimizaciones
        if analysis['optimization']['suggestions']:
            for suggestion in analysis['optimization']['suggestions']:
                if suggestion['priority'] == 'high':
                    recommendations.append(f"💡 {suggestion['suggestion']}")
        
        return recommendations
    
    def _estimate_time_savings(self, suggestions, mesh) -> Dict:
        """Estima el ahorro de tiempo con las optimizaciones"""
        total_savings_percent = 0
        
        for suggestion in suggestions:
            if suggestion['type'] == 'orientation':
                total_savings_percent += 20
            elif suggestion['type'] == 'hollowing':
                total_savings_percent += 40
        
        return {
            'time_savings_percent': total_savings_percent,
            'material_savings_percent': 60 if any(s['type'] == 'hollowing' for s in suggestions) else 0
        }

    refresh_token = create_refresh_token(identity=user.id)

        "refresh_token": refresh_token,

import numpy as np
from stl import mesh
import os

class STLAnalyzer:
    """Analizador real de archivos STL usando numpy-stl"""
    
    @staticmethod
    def analyze(file_path, density_pla=1.24, print_speed=60, layer_height=0.2):
        """
        Analiza un archivo STL y devuelve métricas completas
        """
        if not os.path.exists(file_path):
            return {"error": "Archivo no encontrado"}
        
        try:
            # Cargar mesh
            stl_mesh = mesh.Mesh.from_file(file_path)
            
            # Calcular volumen usando divergencia teorema
            volume = stl_mesh.volume
            volume_cm3 = volume / 1000  # mm³ a cm³
            
            # Área superficial
            surface_area = sum(stl_mesh.areas)
            surface_area_cm2 = surface_area / 100
            
            # Bounding box
            min_point = stl_mesh.min_
            max_point = stl_mesh.max_
            bounding_box = {
                'x': float(max_point[0] - min_point[0]),
                'y': float(max_point[1] - min_point[1]),
                'z': float(max_point[2] - min_point[2])
            }
            
            # Número de triángulos
            triangles = len(stl_mesh.vectors)
            
            # Verificar manifold (simplificado)
            is_manifold = STLAnalyzer._check_manifold(stl_mesh)
            
            # Estimar peso (PLA density 1.24 g/cm³)
            estimated_weight = volume_cm3 * density_pla
            
            # Estimar tiempo de impresión
            estimated_time = STLAnalyzer._estimate_print_time(
                volume_cm3, surface_area_cm2, 
                layer_height, print_speed, bounding_box
            )
            
            # Detectar problemas
            issues = STLAnalyzer._detect_issues(stl_mesh, bounding_box)
            
            # Recomendación de orientación
            orientation = STLAnalyzer._recommend_orientation(bounding_box)
            
            # Soportes necesarios
            support_needed = STLAnalyzer._needs_supports(stl_mesh)
            
            return {
                'success': True,
                'filename': os.path.basename(file_path),
                'volume_cm3': round(volume_cm3, 2),
                'surface_area_cm2': round(surface_area_cm2, 2),
                'bounding_box_mm': bounding_box,
                'triangles': triangles,
                'is_manifold': is_manifold,
                'is_waterproof': is_manifold,
                'estimated_weight_grams': round(estimated_weight, 2),
                'estimated_time_hours': round(estimated_time, 2),
                'issues': issues,
                'orientation_recommendation': orientation,
                'support_needed': support_needed,
                'density_used': density_pla
            }
            
        except Exception as e:
            return {"error": str(e)}
    
    @staticmethod
    def _check_manifold(stl_mesh):
        """Verificación simplificada de manifold"""
        try:
            # Si el volumen es positivo y finito, probablemente es manifold
            return stl_mesh.volume > 0 and np.isfinite(stl_mesh.volume)
        except:
            return False
    
    @staticmethod
    def _estimate_print_time(volume_cm3, surface_area_cm2, layer_height, speed, bbox):
        """
        Estimación de tiempo basada en volumen, capas y velocidad
        Fórmula: (volumen / (speed * layer_height * nozzle_area)) * factor
        """
        nozzle_diameter = 0.4  # mm
        nozzle_area = np.pi * (nozzle_diameter/2)**2  # mm²
        
        # Convertir volumen a mm³
        volume_mm3 = volume_cm3 * 1000
        
        # Tiempo base de extrusión
        extrusion_time = volume_mm3 / (speed * nozzle_area * layer_height * 10)  # en minutos
        
        # Añadir tiempo de movimiento y capas
        num_layers = bbox['z'] / layer_height
        travel_time = num_layers * 2  # 2 segundos por capa de movimiento
        
        total_minutes = extrusion_time + travel_time
        return total_minutes / 60  # Convertir a horas
    
    @staticmethod
    def _detect_issues(stl_mesh, bbox):
        """Detectar problemas comunes en el modelo"""
        issues = []
        
        # Paredes muy finas
        min_dimension = min(bbox['x'], bbox['y'], bbox['z'])
        if min_dimension < 1.0:
            issues.append({
                'type': 'thin_wall',
                'severity': 'high',
                'message': f'Dimensión mínima muy pequeña: {min_dimension:.2f}mm'
            })
        
        # Modelo muy grande
        max_dimension = max(bbox['x'], bbox['y'], bbox['z'])
        if max_dimension > 200:
            issues.append({
                'type': 'large_model',
                'severity': 'medium',
                'message': f'Modelo grande: {max_dimension:.1f}mm. Verificar volumen de impresión.'
            })
        
        # Volumen cero o negativo
        if stl_mesh.volume <= 0:
            issues.append({
                'type': 'invalid_volume',
                'severity': 'high',
                'message': 'Volumen inválido. El modelo puede no ser manifold.'
            })
        
        # Muchos triángulos (modelo muy denso)
        if len(stl_mesh.vectors) > 100000:
            issues.append({
                'type': 'high_poly',
                'severity': 'low',
                'message': f'Modelo con {len(stl_mesh.vectors)} triángulos. Puede ralentizar el slicing.'
            })
        
        return issues
    
    @staticmethod
    def _recommend_orientation(bbox):
        """Recomendar mejor orientación para imprimir"""
        dimensions = {
            'X': bbox['x'],
            'Y': bbox['y'],
            'Z': bbox['z']
        }
        
        # La dimensión más pequeña debe ir en Z para minimizar soportes
        sorted_dims = sorted(dimensions.items(), key=lambda x: x[1])
        
        return {
            'recommended_axis': f'{sorted_dims[0][0]} en vertical (Z)',
            'reason': f'Minimiza soportes. Dimensión más pequeña: {sorted_dims[0][1]:.1f}mm',
            'dimensions_sorted': dimensions
        }
    
    @staticmethod
    def _needs_supports(stl_mesh):
        """Estimar si necesita soportes basado en overhangs"""
        # Simplificación: verificar si hay caras con ángulo > 45°
        try:
            normals = stl_mesh.normals
            # Si hay muchas normales con componente Z negativa significativa
            z_normals = normals[:, 2]
            overhang_ratio = np.sum(z_normals < -0.5) / len(z_normals)
            return overhang_ratio > 0.1
        except:
            return True

class CostCalculator:
    """
    Calculadora profesional de costes de impresión 3D
    Incluye: material, electricidad, depreciación, mano de obra, margen
    """
    
    @staticmethod
    def calculate(
        weight_grams,
        print_time_hours,
        filament_price_kg=80.0,
        electricity_kwh=0.90,
        printer_power_kw=0.250,
        labor_hour=50.0,
        depreciation_rate=0.15,
        printer_price=5000.0,
        printer_lifetime_hours=10000,
        margin_percent=30.0,
        discount_percent=0.0,
        tax_percent=0.0,
        post_processing_hours=0.0,
        support_material_grams=0.0,
        shipping_cost=0.0
    ):
        """
        Calcula el precio final completo
        """
        # 1. Coste de material
        material_cost = (weight_grams / 1000) * filament_price_kg
        support_cost = (support_material_grams / 1000) * filament_price_kg
        total_material = material_cost + support_cost
        
        # 2. Coste de electricidad
        electricity_cost = print_time_hours * printer_power_kw * electricity_kwh
        
        # 3. Depreciación de máquina
        depreciation_per_hour = (printer_price * depreciation_rate) / printer_lifetime_hours
        depreciation_cost = print_time_hours * depreciation_per_hour
        
        # 4. Mano de obra
        labor_cost = (print_time_hours * 0.1 + post_processing_hours) * labor_hour
        
        # 5. Coste total de producción
        production_cost = total_material + electricity_cost + depreciation_cost + labor_cost
        
        # 6. Margen de beneficio
        margin = production_cost * (margin_percent / 100)
        
        # 7. Subtotal
        subtotal = production_cost + margin
        
        # 8. Descuento
        discount = subtotal * (discount_percent / 100)
        after_discount = subtotal - discount
        
        # 9. Impuestos
        tax = after_discount * (tax_percent / 100)
        
        # 10. Total final
        total = after_discount + tax + shipping_cost
        
        # 11. Beneficio neto
        net_profit = margin - discount
        
        return {
            'breakdown': {
                'material_cost': round(total_material, 2),
                'filament_cost': round(material_cost, 2),
                'support_cost': round(support_cost, 2),
                'electricity_cost': round(electricity_cost, 2),
                'depreciation_cost': round(depreciation_cost, 2),
                'labor_cost': round(labor_cost, 2),
                'production_cost': round(production_cost, 2),
                'margin': round(margin, 2),
                'discount': round(discount, 2),
                'tax': round(tax, 2),
                'shipping': round(shipping_cost, 2)
            },
            'pricing': {
                'subtotal': round(subtotal, 2),
                'after_discount': round(after_discount, 2),
                'total': round(total, 2),
                'net_profit': round(net_profit, 2),
                'profit_margin_percent': round((net_profit / total * 100) if total > 0 else 0, 2)
            },
            'summary': {
                'weight_grams': weight_grams,
                'print_time_hours': round(print_time_hours, 2),
                'filament_price_kg': filament_price_kg,
                'margin_percent': margin_percent
            }
        }
    
    @staticmethod
    def compare_technologies(weight_grams, print_time_fdm):
        """
        Compara costes entre FDM, Resina y SLS
        """
        fdm = CostCalculator.calculate(
            weight_grams=weight_grams,
            print_time_hours=print_time_fdm,
            filament_price_kg=80.0,
            printer_power_kw=0.250,
            printer_price=5000.0
        )
        
        # Resina: más cara el material, pero más rápida para detalles
        resin_weight = weight_grams * 1.1  # 10% más por soportes
        resin_time = print_time_fdm * 0.6
        resin = CostCalculator.calculate(
            weight_grams=resin_weight,
            print_time_hours=resin_time,
            filament_price_kg=150.0,  # Resina más cara
            printer_power_kw=0.100,
            printer_price=8000.0
        )
        
        return {
            'fdm': fdm,
            'resin': resin,
            'recommendation': 'FDM' if fdm['pricing']['total'] < resin['pricing']['total'] else 'Resina'
        }
    
    @staticmethod
    def financial_simulator(
        current_monthly_revenue=10000,
        price_increase_percent=10,
        new_printer_cost=5000,
        new_printer_monthly_capacity=3000,
        fixed_costs=2000,
        variable_cost_percent=40
    ):
        """
        Simulador financiero: qué pasa si subo precios, compro máquina, etc.
        """
        # Escenario actual
        current_variable_costs = current_monthly_revenue * (variable_cost_percent / 100)
        current_profit = current_monthly_revenue - current_variable_costs - fixed_costs
        
        # Escenario con subida de precios
        new_revenue = current_monthly_revenue * (1 + price_increase_percent / 100)
        new_variable_costs = new_revenue * (variable_cost_percent / 100)
        new_profit = new_revenue - new_variable_costs - fixed_costs
        
        # Escenario con nueva impresora
        additional_revenue = new_printer_monthly_capacity
        additional_fixed = new_printer_cost / 36  # Amortización 3 años
        scenario_printer_profit = current_profit + additional_revenue * (1 - variable_cost_percent/100) - additional_fixed
        
        # Punto de equilibrio
        break_even = fixed_costs / (1 - variable_cost_percent/100)
        
        # ROI nueva impresora
        monthly_gain = new_printer_monthly_capacity * (1 - variable_cost_percent/100)
        roi_months = new_printer_cost / monthly_gain if monthly_gain > 0 else 0
        
        return {
            'current_scenario': {
                'revenue': current_monthly_revenue,
                'profit': current_profit,
                'margin_percent': round(current_profit / current_monthly_revenue * 100, 2) if current_monthly_revenue > 0 else 0
            },
            'price_increase_scenario': {
                'new_revenue': round(new_revenue, 2),
                'new_profit': round(new_profit, 2),
                'profit_increase': round(new_profit - current_profit, 2),
                'price_increase_percent': price_increase_percent
            },
            'new_printer_scenario': {
                'additional_monthly_revenue': additional_revenue,
                'additional_monthly_profit': round(additional_revenue * (1 - variable_cost_percent/100) - additional_fixed, 2),
                'roi_months': round(roi_months, 1),
                'annual_roi_percent': round((12 / roi_months * 100) if roi_months > 0 else 0, 2)
            },
            'break_even_monthly': round(break_even, 2),
            'recommendations': CostCalculator._generate_recommendations(
                current_profit, new_profit, roi_months, break_even
            )
        }
    
    @staticmethod
    def _generate_recommendations(current_profit, new_profit, roi_months, break_even):
        recs = []
        if new_profit > current_profit * 1.2:
            recs.append("✅ Subir precios un 10% aumentaría significativamente el beneficio")
        if roi_months < 12:
            recs.append(f"✅ Nueva impresora se paga en {roi_months:.0f} meses. Buena inversión.")
        elif roi_months < 24:
            recs.append(f"️ Nueva impresora se paga en {roi_months:.0f} meses. Evaluar demanda.")
        else:
            recs.append(f"❌ Nueva impresora tarda {roi_months:.0f} meses en pagarse. Riesgo alto.")
        recs.append(f"📊 Punto de equilibrio mensual: R$ {break_even:.2f}")
        return recs

from app.models import Order, Invoice, Client, InventoryItem, Printer
from datetime import datetime, timedelta
from sqlalchemy import func

class DashboardAnalytics:
    """Analytics completo para dashboard"""
    
    @staticmethod
    def get_kpis():
        """Obtiene KPIs principales"""
        now = datetime.utcnow()
        last_30_days = now - timedelta(days=30)
        
        # Ingresos del mes
        monthly_invoices = Invoice.query.filter(
            Invoice.status == 'paid',
            Invoice.issue_date >= last_30_days
        ).all()
        monthly_revenue = sum(i.total for i in monthly_invoices)
        
        # Pedidos activos
        active_orders = Order.query.filter(
            Order.status.in_(['pending', 'printing', 'finishing'])
        ).count()
        
        # Horas impresas (estimado)
        printing_orders = Order.query.filter(
            Order.status == 'printing',
            Order.estimated_hours.isnot(None)
        ).all()
        total_printing_hours = sum(o.estimated_hours or 0 for o in printing_orders)
        
        # Material consumido (estimado)
        total_weight = sum(
            sum(item.get('weight_grams', 0) for item in (o.items or []))
            for o in Order.query.filter(Order.status == 'delivered').all()
        )
        
        # Clientes activos
        active_clients = Client.query.filter(
            Client.status == 'customer'
        ).count()
        
        # Beneficio (estimado 30% margen)
        estimated_profit = monthly_revenue * 0.30
        
        return {
            'monthly_revenue': round(monthly_revenue, 2),
            'active_orders': active_orders,
            'printing_hours': round(total_printing_hours, 2),
            'material_consumed_grams': round(total_weight, 2),
            'active_clients': active_clients,
            'estimated_profit': round(estimated_profit, 2),
            'period': 'last_30_days'
        }
    
    @staticmethod
    def get_revenue_chart(months=6):
        """Datos para gráfico de ingresos"""
        now = datetime.utcnow()
        data = []
        
        for i in range(months):
            month_start = now - timedelta(days=30*(i+1))
            month_end = now - timedelta(days=30*i)
            
            invoices = Invoice.query.filter(
                Invoice.status == 'paid',
                Invoice.issue_date >= month_start,
                Invoice.issue_date < month_end
            ).all()
            
            revenue = sum(inv.total for inv in invoices)
            data.append({
                'month': month_start.strftime('%b %Y'),
                'revenue': round(revenue, 2)
            })
        
        return list(reversed(data))
    
    @staticmethod
    def get_orders_by_status():
        """Distribución de pedidos por estado"""
        statuses = ['pending', 'printing', 'finishing', 'shipped', 'delivered', 'cancelled']
        data = []
        
        for status in statuses:
            count = Order.query.filter_by(status=status).count()
            if count > 0:
                data.append({'status': status, 'count': count})
        
        return data
    
    @staticmethod
    def get_top_clients(limit=10):
        """Top clientes por gasto"""
        clients = Client.query.order_by(Client.total_spent.desc()).limit(limit).all()
        return [
            {
                'name': c.contact_name,
                'company': c.company_name,
                'total_spent': round(c.total_spent, 2),
                'orders': c.orders.count()
            }
            for c in clients
        ]
    
    @staticmethod
    def get_inventory_alerts():
        """Alertas de inventario bajo"""
        items = InventoryItem.query.filter(
            InventoryItem.quantity <= InventoryItem.min_stock,
            InventoryItem.alert_enabled == True
        ).all()
        
        return [
            {
                'sku': item.sku,
                'name': item.name,
                'quantity': item.quantity,
                'min_stock': item.min_stock,

            return {'message': 'No hay datos suficientes'}

        total_cost = total_revenue * 0.70  # Estimado
        total_profit = total_revenue - total_cost
        
        return {
            'total_revenue': round(total_revenue, 2),
            'total_cost': round(total_cost, 2),
            'total_profit': round(total_profit, 2),
            'margin_percent': round((total_profit / total_revenue * 100) if total_revenue > 0 else 0, 2),
            'avg_order_value': round(total_revenue / len(delivered_orders), 2),
            'total_orders': len(delivered_orders)
        }

def add_movement():
    data = request.get_json()
    user_id = get_jwt_identity()
    
    item = InventoryItem.query.get_or_404(data['item_id'])
    
    movement = StockMovement(
        item_id=data['item_id'],
        type=data['type'],  # in, out, adjustment
        quantity=data['quantity'],
        reason=data.get('reason'),
        reference=data.get('reference'),
        user_id=user_id
    )
    
    # Actualizar stock
    if data['type'] == 'in':
        item.quantity += data['quantity']
    elif data['type'] == 'out':
        item.quantity -= data['quantity']
    elif data['type'] == 'adjustment':
        item.quantity = data['quantity']
    
    db.session.add(movement)
    db.session.commit()
    
    return jsonify({"movement": movement.to_dict(), "new_quantity": item.quantity}), 201

import { create } from 'zustand'
import { authAPI } from '@/services/api'

export const useAuthStore = create((set) => ({
  user: null,
  token: localStorage.getItem('fazluiz_token'),
  isAuthenticated: !!localStorage.getItem('fazluiz_token'),
  loading: false,
  
  login: async (email, password) => {
    set({ loading: true })
    try {
      const { data } = await authAPI.login({ email, password })
      localStorage.setItem('fazluiz_token', data.access_token)
      set({ 
        user: data.user, 
        token: data.access_token, 
        isAuthenticated: true,
        loading: false 
      })
      return { success: true }
    } catch (error) {
      set({ loading: false })
      return { success: false, error: error.response?.data?.error }
    }
  },
  
  logout: () => {
    localStorage.removeItem('fazluiz_token')
    set({ user: null, token: null, isAuthenticated: false })
  },
  
  loadUser: async () => {
    try {
      const { data } = await authAPI.me()
      set({ user: data })
    } catch (error) {
      console.error('Error loading user:', error)
    }
  }
}))

export function STLAnalyzer() {
  const [file, setFile] = useState(null)
  const [previewUrl, setPreviewUrl] = useState(null)
  const [analysis, setAnalysis] = useState(null)
  const [loading, setLoading] = useState(false)
  const fileInputRef = useRef(null)
  
  const handleFileChange = (e) => {
    const selected = e.target.files[0]
    if (selected && selected.name.toLowerCase().endsWith('.stl')) {
      setFile(selected)
      setPreviewUrl(URL.createObjectURL(selected))
      setAnalysis(null)
    }
  }
  
  const handleAnalyze = async () => {
    if (!file) return
    setLoading(true)
    try {
      const { data } = await stlAPI.analyze(file)
      setAnalysis(data.result)
      toast.success('Análisis completado')
    } catch (error) {
      toast.error('Error al analizar STL')
    } finally {
      setLoading(false)
    }

function PrinterModel({ position, printer, onClick }) {
  const meshRef = useRef()
  const [hovered, setHovered] = useState(false)
  
  const statusColors = {

    printing: '#10B981',

  useFrame((state) => {

"""
TIBERION CORE — Orquestador central multi-agente
Arquitectura inspirada en LangGraph + CrewAI
"""
import os
import json
from typing import Dict, List, Any, Optional
from datetime import datetime
from dataclasses import dataclass
from enum import Enum

from openai import OpenAI

class AgentType(Enum):
    CORE = "core"
    FINANCE = "finance"
    PRINT = "print"
    SALES = "sales"
    INVENTORY = "inventory"
    MAINTENANCE = "maintenance"
    CEO = "ceo"
    VISION = "vision"

@dataclass
class Memory:
    """Memoria persistente del sistema"""
    short_term: List[Dict] = []  # Últimas conversaciones
    long_term: Dict = {}         # Conocimiento del negocio
    episodic: List[Dict] = []    # Eventos importantes
    
    def add_short_term(self, role: str, content: str):
        self.short_term.append({
            "role": role,
            "content": content,
            "timestamp": datetime.utcnow().isoformat()
        })
        # Mantener solo últimas 50 interacciones
        if len(self.short_term) > 50:
            self.short_term = self.short_term[-50:]
    
    def add_long_term(self, key: str, value: Any):
        self.long_term[key] = value
    
    def add_episodic(self, event_type: str, data: Dict):
        self.episodic.append({
            "type": event_type,
            "data": data,
            "timestamp": datetime.utcnow().isoformat()
        })

class TiberionCore:
    """
    Núcleo central de Tiberion
    Coordina todos los agentes especializados
    """
    
    def __init__(self, app_context=None):
        self.app_context = app_context
        self.client = OpenAI(api_key=os.environ.get('OPENAI_API_KEY'))
        self.memory = Memory()
        self.agents = {}
        self._register_agents()
    
    def _register_agents(self):
        """Registra todos los agentes especializados"""
        from tiberion.agents.finance import FinanceAgent
        from tiberion.agents.print_agent import PrintAgent
        from tiberion.agents.sales import SalesAgent
        from tiberion.agents.ceo import CEOAgent
        
        self.agents[AgentType.FINANCE] = FinanceAgent(self)
        self.agents[AgentType.PRINT] = PrintAgent(self)
        self.agents[AgentType.SALES] = SalesAgent(self)
        self.agents[AgentType.CEO] = CEOAgent(self)
    
    def classify_intent(self, message: str) -> AgentType:
        """
        Clasifica la intención del usuario para delegar al agente correcto
        """
        prompt = f"""
Eres un clasificador de intenciones para un asistente de negocio de impresión 3D.

Mensaje del usuario: "{message}"

Clasifica en UNA de estas categorías:
- finance: preguntas sobre ingresos, gastos, beneficios, facturas, precios
- print: preguntas sobre impresoras, STL, materiales, producción, impresión
- sales: preguntas sobre clientes, leads, presupuestos, ventas
- ceo: informes generales, resúmenes, toma de decisiones estratégicas
- core: saludos, preguntas generales, otras

Responde SOLO con la categoría (finance, print, sales, ceo o core).
"""
        try:
            response = self.client.chat.completions.create(
                model="gpt-4o-mini",
                messages=[{"role": "user", "content": prompt}],
                temperature=0.1,
                max_tokens=10
            )
            category = response.choices[0].message.content.strip().lower()
            mapping = {
                'finance': AgentType.FINANCE,
                'print': AgentType.PRINT,
                'sales': AgentType.SALES,
                'ceo': AgentType.CEO
            }
            return mapping.get(category, AgentType.CORE)
        except Exception:
            return AgentType.CORE
    
    def process(self, message: str, context: Optional[Dict] = None) -> Dict:
        """
        Procesa un mensaje del usuario
        """
        # Guardar en memoria
        self.memory.add_short_term("user", message)
        
        # Clasificar intención
        agent_type = self.classify_intent(message)
        
        # Delegar al agente correspondiente
        if agent_type in self.agents:
            response = self.agents[agent_type].process(message, context)
        else:
            response = self._handle_core(message, context)
        
        # Guardar respuesta en memoria
        self.memory.add_short_term("assistant", response['content'])
        
        return response
    
    def _handle_core(self, message: str, context: Optional[Dict]) -> Dict:
        """Maneja mensajes generales"""
        system_prompt = """
Eres Tiberion, el asistente IA de FazLuiz 3D, una empresa de impresión 3D profesional 
propiedad de Thiago Luiz en Brasil.

Tus capacidades:
- Gestionar CRM, pedidos, inventario, facturación
- Analizar archivos STL
- Calcular costes de impresión
- Generar presupuestos
- Analizar rentabilidad del negocio
- Recomendar acciones estratégicas

Responde de forma clara, profesional y útil. En portugués brasileño cuando sea apropiado.
"""
        
        messages = [
            {"role": "system", "content": system_prompt},
            *self.memory.short_term[-10:]  # Últimas 10 interacciones
        ]
        
        try:
            response = self.client.chat.completions.create(
                model="gpt-4o",
                messages=messages,
                temperature=0.7,
                max_tokens=1000
            )
            content = response.choices[0].message.content
            return {
                "content": content,
                "agent": "core",
                "actions": [],
                "timestamp": datetime.utcnow().isoformat()
            }
        except Exception as e:
            return {
                "content": f"Disculpa, tuve un problema: {str(e)}",
                "agent": "core",
                "actions": []
            }
    
    def ceo_morning_report(self) -> Dict:
        """Genera informe matutino para el CEO"""
        return self.agents[AgentType.CEO].generate_morning_report()

from datetime import datetime, timedelta
from typing import Dict, Optional
from app.models import Order, Invoice, Client
from app.extensions import db
from sqlalchemy import func

class CEOAgent:
    def __init__(self, core):
        self.core = core
    
    def process(self, message: str, context: Optional[Dict]) -> Dict:
        """Procesa preguntas del CEO"""
        # Detectar tipo de pregunta
        if any(w in message.lower() for w in ['informe', 'report', 'ceo', 'mañana', 'hoy']):
            return self.generate_morning_report()
        elif 'ingres' in message.lower() or 'gan' in message.lower():
            return self._revenue_query(message)
        elif 'cliente' in message.lower() and ('mejor' in message.lower() or 'top' in message.lower()):
            return self._top_clients_query()
        else:
            return self._general_query(message)
    
    def generate_morning_report(self) -> Dict:
        """Informe matutino completo"""
        today = datetime.utcnow().date()
        yesterday = today - timedelta(days=1)
        last_30 = datetime.utcnow() - timedelta(days=30)
        
        # Métricas
        yesterday_invoices = Invoice.query.filter(
            func.date(Invoice.issue_date) == yesterday,
            Invoice.status == 'paid'
        ).all()
        yesterday_revenue = sum(i.total for i in yesterday_invoices)
        
        pending_quotes = db.session.query(func.count(Quote.id)).filter(
            Quote.status.in_(['sent', 'draft'])
        ).scalar()
        
        active_orders = Order.query.filter(
            Order.status.in_(['pending', 'printing', 'finishing'])
        ).count()
        
        new_clients = Client.query.filter(
            func.date(Client.created_at) == yesterday
        ).count()
        
        low_stock_items = db.session.query(func.count(InventoryItem.id)).filter(
            InventoryItem.quantity <= InventoryItem.min_stock
        ).scalar()
        
        # Generar narrativa con IA
        narrative = self._generate_narrative({
            'yesterday_revenue': yesterday_revenue,
            'pending_quotes': pending_quotes,
            'active_orders': active_orders,
            'new_clients': new_clients,
            'low_stock': low_stock_items
        })
        
        return {
            "content": narrative,
            "agent": "ceo",
            "actions": [],
            "metrics": {
                "yesterday_revenue": yesterday_revenue,
                "pending_quotes": pending_quotes,
                "active_orders": active_orders,
                "new_clients": new_clients,
                "low_stock_items": low_stock_items
            }
        }
    
    def _generate_narrative(self, metrics: Dict) -> str:
        """Genera narrativa ejecutiva"""
        prompt = f"""
Genera un informe ejecutivo breve y directo para Thiago Luiz, dueño de FazLuiz 3D.

Métricas de hoy:
- Facturación ayer: R$ {metrics['yesterday_revenue']:.2f}

        if not printing:
            return {
                "content": "No hay pedidos imprimiéndose ahora mismo.",
                "agent": "print",
                "actions": []
            }
        
        # El que tenga menos tiempo restante (estimado)
        next_one = min(printing, key=lambda o: o.estimated_hours or 999)
        return {
            "content": f"🖨️ La impresora que termina antes está trabajando en el pedido {next_one.order_number}, con {next_one.estimated_hours or '?'} horas estimadas.",
            "agent": "print",
            "actions": []
        }

            return {
                "content": f"✅ Impresoras disponibles ahora: {names}",
                "agent": "print",
                "actions": []
            }

            "content": "Todas las impresoras están ocupadas. La que antes queda libre es...",

"""
Event Bus — Comunicación desacoplada entre módulos
Permite que los módulos reaccionen a eventos sin depender unos de otros
"""
from typing import Callable, Dict, List, Any
from dataclasses import dataclass
from datetime import datetime
import threading

@dataclass
class Event:
    type: str
    data: Dict[str, Any]
    timestamp: datetime
    source: str

class EventBus:
    """
    Bus de eventos central
    Permite comunicación desacoplada entre módulos
    """
    
    _instance = None
    _lock = threading.Lock()
    
    def __new__(cls):
        if cls._instance is None:
            with cls._lock:
                if cls._instance is None:
                    cls._instance = super().__new__(cls)
                    cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if self._initialized:
            return
        self._initialized = True
        self.subscribers: Dict[str, List[Callable]] = {}
        self.history: List[Event] = []
    
    def subscribe(self, event_type: str, handler: Callable):
        """Suscribe un handler a un tipo de evento"""
        if event_type not in self.subscribers:
            self.subscribers[event_type] = []
        self.subscribers[event_type].append(handler)
    
    def publish(self, event_type: str, data: Dict[str, Any], source: str = "system"):
        """Publica un evento"""
        event = Event(
            type=event_type,
            data=data,
            timestamp=datetime.utcnow(),
            source=source
        )
        self.history.append(event)
        
        # Mantener solo últimos 1000 eventos
        if len(self.history) > 1000:
            self.history = self.history[-1000:]
        
        # Notificar suscriptores
        for handler in self.subscribers.get(event_type, []):
            try:
                handler(event)
            except Exception as e:
                print(f"Error en handler de {event_type}: {e}")
    
    def get_history(self, event_type: str = None, limit: int = 100) -> List[Event]:
        """Obtiene historial de eventos"""
        if event_type:
            events = [e for e in self.history if e.type == event_type]
        else:
            events = self.history
        return events[-limit:]

            return {
                'success': True,

                'recommendations': PrintFailureDetector._generate_recommendations(issues)

        height = img_array.shape[0]
        # Buscar la parte superior de la pieza
        non_empty_rows = np.where(np.sum(img_array, axis=(1, 2)) > 0)[0]

            return 0.0

"""

logger = logging.getLogger(__name__)

class VoiceSystem:
    """
    Sistema de voz bidireccional
    """
    
    def __init__(self, language: str = 'es-ES', voice_name: str = 'Jarvis'):
        self.language = language
        self.voice_name = voice_name
        self.recognizer = sr.Recognizer()
        self.microphone = sr.Microphone()
        
        # Configurar recognizer
        self.recognizer.energy_threshold = 300
        self.recognizer.dynamic_energy_threshold = True
        
        # Callbacks
        self.on_speech_detected: Optional[Callable] = None
        self.on_transcription: Optional[Callable] = None
        
        # Estado
        self.is_listening = False
        self._listening_thread = None
        
        logger.info(f"🎤 Sistema de voz inicializado ({language})")
    
    def speak(self, text: str, blocking: bool = True):
        """
        Convierte texto a voz y reproduce
        """
        logger.info(f"🔊 Speaking: {text}")
        
        try:
            # Crear TTS
            tts = gTTS(text=text, lang=self.language, slow=False)
            
            # Guardar en archivo temporal
            with tempfile.NamedTemporaryFile(delete=False, suffix='.mp3') as fp:
                temp_path = fp.name
            
            tts.save(temp_path)
            
            # Reproducir
            if blocking:
                playsound(temp_path)
            else:
                threading.Thread(target=playsound, args=(temp_path,)).start()
            
            # Limpiar
            os.unlink(temp_path)
            
        except Exception as e:
            logger.error(f"Error en TTS: {e}")
    
    def listen_once(self, timeout: int = 5) -> Optional[str]:
        """
        Escucha una vez y retorna el texto
        """
        with self.microphone as source:
            logger.info("👂 Escuchando...")
            self.recognizer.adjust_for_ambient_noise(source, duration=0.5)
            
            try:
                audio = self.recognizer.listen(source, timeout=timeout)
                logger.info("🎵 Audio capturado, procesando...")
                
                # Usar Google Speech Recognition
                text = self.recognizer.recognize_google(audio, language=self.language)
                logger.info(f"📝 Transcrito: {text}")
                
                if self.on_transcription:
                    self.on_transcription(text)
                
                return text
                
            except sr.WaitTimeoutError:
                logger.warning("⏱️ Timeout esperando voz")
            except sr.UnknownValueError:
                logger.warning("❓ No se entendió el audio")
            except sr.RequestError as e:
                logger.error(f"❌ Error en servicio de reconocimiento: {e}")
        
        return None
    
    def start_continuous_listening(self, callback: Callable[[str], None]):
        """
        Escucha continua en background
        """
        self.is_listening = True
        
        def _listen_loop():
            while self.is_listening:

        self._listening_thread = threading.Thread(target=_listen_loop, daemon=True)

"""
Sistema de memoria para JARVIS
- Corto plazo: conversaciones recientes
- Largo plazo: preferencias, datos importantes
- Vectorial: búsqueda semántica
"""
import os
import json
import logging
from datetime import datetime, timedelta
from typing import List, Dict, Optional
from dataclasses import dataclass, asdict
import chromadb
from chromadb.config import Settings

logger = logging.getLogger(__name__)

@dataclass
class MemoryEntry:
    timestamp: str
    role: str  # 'user' o 'assistant'
    content: str
    metadata: Dict = None

class MemorySystem:
    """
    Sistema de memoria híbrido
    """
    
    def __init__(self, persist_dir: str = './jarvis_memory'):
        self.persist_dir = persist_dir
        os.makedirs(persist_dir, exist_ok=True)
        
        # Memoria conversacional (corto plazo)
        self.conversation_file = os.path.join(persist_dir, 'conversation.json')
        self.conversation: List[MemoryEntry] = self._load_conversation()
        
        # Memoria vectorial (largo plazo)
        self.client = chromadb.Client(Settings(
            persist_directory=persist_dir,
            anonymized_telemetry=False
        ))
        
        # Colecciones
        self.long_term_memory = self.client.get_or_create_collection(
            name="long_term_memory",
            metadata={"description": "Memoria a largo plazo"}
        )
        
        self.preferences = self.client.get_or_create_collection(
            name="preferences",
            metadata={"description": "Preferencias del usuario"}
        )
        
        self.knowledge = self.client.get_or_create_collection(
            name="knowledge",
            metadata={"description": "Conocimiento general"}
        )
        
        logger.info("🧠 Sistema de memoria inicializado")
    
    def _load_conversation(self) -> List[MemoryEntry]:
        """Carga conversación desde archivo"""
        if os.path.exists(self.conversation_file):
            try:
                with open(self.conversation_file, 'r') as f:
                    data = json.load(f)
                    return [MemoryEntry(**entry) for entry in data]
            except:
                pass
        return []
    
    def _save_conversation(self):
        """Guarda conversación en archivo"""
        with open(self.conversation_file, 'w') as f:
            json.dump([asdict(entry) for entry in self.conversation], f, indent=2)
    
    def add_interaction(self, role: str, content: str, metadata: Dict = None):
        """Agrega interacción a memoria de corto plazo"""
        entry = MemoryEntry(
            timestamp=datetime.now().isoformat(),
            role=role,
            content=content,
            metadata=metadata or {}
        )
        
        self.conversation.append(entry)
        
        # Mantener solo últimas 100 interacciones
        if len(self.conversation) > 100:
            self.conversation = self.conversation[-100:]
        
        self._save_conversation()
        
        # También guardar en memoria vectorial si es importante
        self._add_to_long_term(role, content, metadata)
    
    def _add_to_long_term(self, role: str, content: str, metadata: Dict = None):
        """Agrega a memoria vectorial"""
        # Solo guardar si parece importante
        if self._is_important(content):
            doc_id = f"{role}_{datetime.now().timestamp()}"
            
            self.long_term_memory.add(
                documents=[content],
                metadatas=[{
                    'role': role,
                    'timestamp': datetime.now().isoformat(),
                    **(metadata or {})
                }],
                ids=[doc_id]
            )
    
    def _is_important(self, content: str) -> bool:
        """Determina si el contenido es importante para guardar"""
        important_keywords = [
            'prefiero', 'me gusta', 'no me gusta',
            'recuerda', 'importante', 'siempre', 'nunca',
            'nombre', 'dirección', 'teléfono', 'email',
            'contraseña', 'clave', 'cuenta'
        ]
        
        content_lower = content.lower()
        return any(kw in content_lower for kw in important_keywords)
    
    def get_recent_interactions(self, limit: int = 10) -> str:
        """Obtiene interacciones recientes formateadas"""
        recent = self.conversation[-limit:]
        
        formatted = []
        for entry in recent:
            role_emoji = "👤" if entry.role == "user" else "🤖"
            formatted.append(f"{role_emoji} {entry.content}")
        
        return "\n".join(formatted)
    
    def search_memory(self, query: str, limit: int = 5) -> List[Dict]:
        """Busca en memoria vectorial"""
        results = self.long_term_memory.query(
            query_texts=[query],
            n_results=limit
        )
        
        return [{
            'content': doc,
            'metadata': meta,
            'distance': dist
        } for doc, meta, dist in zip(
            results['documents'][0],
            results['metadatas'][0],
            results['distances'][0]
        )]
    
    def save_preference(self, category: str, key: str, value: any):
        """Guarda preferencia del usuario"""
        doc_id = f"pref_{category}_{key}"
        
        self.preferences.upsert(
            documents=[f"{category}.{key}: {value}"],
            metadatas={
                'category': category,
                'key': key,
                'value': value,
                'updated_at': datetime.now().isoformat()
            },
            ids=[doc_id]
        )
        
        logger.info(f"💾 Preferencia guardada: {category}.{key}")
    
    def get_preference(self, category: str, key: str = None) -> any:
        """Obtiene preferencia"""
        if key:
            doc_id = f"pref_{category}_{key}"
            results = self.preferences.get(ids=[doc_id])
            if results['metadatas']:
                return results['metadatas'][0].get('value')
        else:
            # Todas las preferencias de una categoría
            results = self.preferences.get(
                where={'category': category}
            )
            return {m['key']: m['value'] for m in results['metadatas']}
        
        return None
    
    def add_reminder(self, text: str, when: datetime):
        """Agrega recordatorio"""
        reminder = {

"""

from typing import Dict

@dataclass

    verbosity: str  # 'concise', 'normal', 'detailed'
    proactivity: float  # 0.0 a 1.0
    humor_level: float  # 0.0 a 1.0

class PersonalityEngine:
    """
    Gestiona la personalidad de JARVIS
    """
    
    def __init__(self, name: str = "Jarvis"):
        self.name = name
        self.config = PersonalityConfig(
            name=name,
            tone='professional',
            verbosity='normal',
            proactivity=0.7,
            humor_level=0.3
        )
        
        self.system_prompts = {
            'professional': self._get_professional_prompt(),
            'casual': self._get_casual_prompt(),
            'humorous': self._get_humorous_prompt()
        }
    
    def get_system_prompt(self) -> str:
        """Obtiene prompt de sistema según configuración"""
        base = self.system_prompts.get(self.config.tone, self._get_professional_prompt())
        
        return f"""
{base}

Configuración:
- Nombre: {self.name}
- Verbosidad: {self.config.verbosity}
- Proactividad: {self.config.proactivity * 100:.0f}%
- Humor: {self.config.humor_level * 100:.0f}%

Instrucciones específicas:
1. Responde de manera {self.config.verbosity}
2. {'Sé proactivo y anticipa necesidades' if self.config.proactivity > 0.5 else 'Responde solo a lo preguntado'}
3. {'Incorpora humor sutil cuando sea apropiado' if self.config.humor_level > 0.3 else 'Mantén un tono profesional'}
4. Siempre sé útil, preciso y respetuoso
5. Si no sabes algo, admítelo y ofrece buscar la información
"""
    
    def _get_professional_prompt(self) -> str:
        return f"""
Eres {self.name}, un asistente de IA profesional y altamente competente.

Características:
- Respuestas precisas y bien estructuradas
- Lenguaje formal pero accesible
- Enfoque en eficiencia y resultados
- Proactivo en sugerir mejoras
- Respetuoso y cortés en todo momento
"""
    
    def _get_casual_prompt(self) -> str:
        return f"""
Eres {self.name}, un asistente de IA amigable y cercano.

Características:
- Tono conversacional y relajado
- Usas expresiones cotidianas
- Empático y comprensivo
- Te adaptas al estado de ánimo del usuario
- Amigable pero profesional cuando es necesario
"""
    
    def _get_humorous_prompt(self) -> str:
        return f"""
Eres {self.name}, un asistente de IA con sentido del humor.

Características:
- Incorporas humor inteligente y sutil
- Haces referencias culturales apropiadas
- Usas ironía ligera cuando corresponde
- Mantienes el profesionalismo pero con diversión
- Evitas humor ofensivo o inapropiado
"""
    
    def adjust_tone(self, tone: str):
        """Ajusta el tono"""
        if tone in self.system_prompts:
            self.config.tone = tone
    
    def set_proactivity(self, level: float):
        """Ajusta nivel de proactividad (0.0 - 1.0)"""
        self.config.proactivity = max(0.0, min(1.0, level))
    
    def learn_style_preference(self, user_feedback: str):
        """Aprende preferencias de estilo del usuario"""
        feedback_lower = user_feedback.lower()
        
        if any(w in feedback_lower for w in ['demasiado formal', 'muy serio']):
            self.config.tone = 'casual'
        elif any(w in feedback_lower for w in ['poco profesional', 'muy informal']):
            self.config.tone = 'professional'
        elif any(w in feedback_lower for w in ['más divertido', 'aburrido']):
            self.config.humor_level = min(1.0, self.config.humor_level + 0.1)
        
        # Guardar preferencia
        from .memory import MemorySystem
        memory = MemorySystem()
        memory.save_preference('personality', 'tone', self.config.tone)
        memory.save_preference('personality', 'humor_level', self.config.humor_level)

"""
JARVIS - Just A Rather Very Intelligent System
Aplicación principal
"""
import os
import sys
import asyncio
import logging
from datetime import datetime
import schedule
import time
import threading

from jarvis.core import JarvisCore, JarvisResponse
from jarvis.voice import VoiceSystem
from jarvis.memory import MemorySystem

# Configurar logging
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.StreamHandler(),
        logging.FileHandler('jarvis.log')
    ]
)

logger = logging.getLogger(__name__)

class Jarvis:
    """
    JARVIS - Asistente IA completo
    """
    
    def __init__(self):
        logger.info("🚀 Inicializando JARVIS...")
        
        # Componentes
        self.core = JarvisCore()
        self.voice = VoiceSystem(language='es-ES')
        self.memory = MemorySystem()
        
        # Estado
        self.is_running = False
        self.wake_word_active = False
        
        # Configurar callbacks
        self.voice.on_transcription = self._on_voice_input
        
        logger.info("✅ JARVIS listo")
    
    def _on_voice_input(self, text: str):
        """Callback cuando se detecta voz"""
        logger.info(f"🎤 Input de voz: {text}")
        
        # Procesar
        response = self.core.process(text)
        
        # Responder por voz
        self.voice.speak(response.text, blocking=False)
    
    def start(self, mode: str = 'voice'):
        """
        Inicia JARVIS
        Modes: 'voice', 'chat', 'hybrid'
        """
        logger.info(f"🎬 Iniciando JARVIS en modo: {mode}")
        self.is_running = True
        
        # Saludo inicial
        greeting = self.core.process("Buenos días")
        self.voice.speak(greeting.text)
        
        if mode == 'voice':
            self._run_voice_mode()
        elif mode == 'chat':
            self._run_chat_mode()
        elif mode == 'hybrid':
            self._run_hybrid_mode()
    
    def _run_voice_mode(self):
        """Modo solo voz"""
        logger.info("🎧 Modo voz activado")
        
        # Escucha continua
        self.voice.start_continuous_listening(self._on_voice_input)
        
        # Tareas programadas
        self._schedule_tasks()
        
        # Mantener ejecutando
        try:
            while self.is_running:
                time.sleep(1)
        except KeyboardInterrupt:
            logger.info("⏹️ Deteniendo JARVIS...")
            self.voice.stop_continuous_listening()
            self.is_running = False
    
    def _run_chat_mode(self):
        """Modo solo texto/consola"""
        logger.info("💬 Modo chat activado")
        
        print(f"\n{'='*60}")
        print(f"  {self.core.personality.name} - Modo Chat")
        print(f"{'='*60}\n")
        print("Escribe 'salir' para terminar\n")
        
        while self.is_running:
            try:
                user_input = input("👤 Tú: ").strip()
                
                if not user_input:
                    continue
                
                if user_input.lower() in ['salir', 'exit', 'quit']:
                    break
                
                # Procesar
                response = self.core.process(user_input)
                
                # Mostrar respuesta
                print(f"\n🤖 {self.core.personality.name}: {response.text}\n")
                
                # Opcional: hablar
                if response.confidence > 0.8:
                    self.voice.speak(response.text, blocking=False)
                
            except KeyboardInterrupt:
                break
            except Exception as e:
                logger.error(f"Error: {e}")
        
        self.is_running = False
    
    def _run_hybrid_mode(self):
        """Modo híbrido: voz + chat"""
        logger.info("🔄 Modo híbrido activado")
        
        # Iniciar escucha de voz en background
        self.voice.start_continuous_listening(self._on_voice_input)
        
        # Ejecutar chat en foreground
        self._run_chat_mode()
        
        # Limpiar
        self.voice.stop_continuous_listening()
    
    def _schedule_tasks(self):
        """Programa tareas periódicas"""
        # Verificaciones proactivas cada hora
        schedule.every().hour.do(self._proactive_check)
        
        # Resumen matutino a las 8 AM
        schedule.every().day.at("08:00").do(self._morning_briefing)
        
        # Resumen nocturno a las 10 PM
        schedule.every().day.at("22:00").do(self._evening_summary)
        
        def run_scheduler():
            while self.is_running:
                schedule.run_pending()
                time.sleep(1)
        
        scheduler_thread = threading.Thread(target=run_scheduler, daemon=True)
        scheduler_thread.start()
    
    def _proactive_check(self):
        """Verificaciones proactivas"""
        suggestions = self.core.proactive_check()
        
        if suggestions:
            for suggestion in suggestions:
                logger.info(f"💡 Proactivo: {suggestion}")
                # Opcional: notificar al usuario
                # self.voice.speak(suggestion)
    
    def _morning_briefing(self):
        """Informe matutino"""
        logger.info("🌅 Generando informe matutino...")
        
        # Obtener informe del core
        report = self.core.process("Dame un resumen de hoy: calendario, clima, tareas pendientes")
        
        # Hablar
        self.voice.speak(f"Buenos días. {report.text}")
    
    def _evening_summary(self):
        """Resumen nocturno"""
        logger.info("🌙 Generando resumen nocturno...")
        
        summary = self.core.process("¿Qué hicimos hoy? Resume las actividades principales")
        
        self.voice.speak(f"Resumen del día: {summary.text}")
    
    def stop(self):
        """Detiene JARVIS"""
        logger.info("🛑 Deteniendo JARVIS...")
        self.is_running = False
        self.voice.stop_continuous_listening()
        print("👋 Hasta luego")


def main():
    """Punto de entrada"""
    import argparse
    
    parser = argparse.ArgumentParser(description='JARVIS - Asistente IA')
    parser.add_argument(
        '--mode', 
        choices=['voice', 'chat', 'hybrid'],
        default='chat',
        help='Modo de operación'
    )
    parser.add_argument(
        '--name',
        default='Jarvis',
        help='Nombre del asistente'
    )
    
    args = parser.parse_args()
    
    # Crear y ejecutar
    jarvis = Jarvis()
    
    try:
        jarvis.start(mode=args.mode)
    except KeyboardInterrupt:
        jarvis.stop()
    except Exception as e:
        logger.error(f"Error fatal: {e}", exc_info=True)
        sys.exit(1)


if __name__ == '__main__':
    main()
